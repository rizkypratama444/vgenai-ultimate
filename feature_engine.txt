  const endpoint=process.env.WEB_SEARCH_URL;
  if(endpoint){
    const u=new URL(endpoint);u.searchParams.set('q',q);
    const r=await withTimeout(fetch(u,{headers:{'accept':'application/json','user-agent':'VickyyVallBot/4.0'}}),FEATURE_LIMITS.searchTimeoutMs);
    if(!r.ok)throw new Error(`Search HTTP ${r.status}`);return normalizeSearchResults(await r.json()).slice(0,limit);
  }
  const r=await withTimeout(fetch(`https://html.duckduckgo.com/html/?q=${encodeURIComponent(q)}`,{headers:{'user-agent':'Mozilla/5.0 VickyyVallBot/4.0','accept-language':'id-ID,id;q=0.9,en;q=0.8'}}),FEATURE_LIMITS.searchTimeoutMs);
  if(!r.ok)throw new Error(`Search HTTP ${r.status}`);
  const html=await r.text(),out=[],re=/<a[^>]+class=["']result__a["'][^>]+href=["']([^"']+)["'][^>]*>([\s\S]*?)<\/a>/gi;
  let m;while((m=re.exec(html))&&out.length<limit){let href=m[1];try{href=decodeURIComponent(href.replace(/^.*?uddg=/,'').split('&')[0]);}catch{}if(/^https?:\/\//i.test(href))out.push({title:stripTags(m[2]),url:href,snippet:''});}

async function downloadRemoteImage(url){
  if(!/^https?:\/\//i.test(url))throw new Error('URL gambar tidak valid.');
  const r=await withTimeout(fetch(url,{redirect:'follow',headers:{'user-agent':'Mozilla/5.0 VickyyVallBot/4.0','accept':'image/avif,image/webp,image/apng,image/*,*/*;q=0.8'}}),10000);
  if(!r.ok)throw new Error(`Gambar HTTP ${r.status}`);
  const type=(r.headers.get('content-type')||'').toLowerCase();
  if(type && !type.startsWith('image/'))throw new Error('URL bukan file gambar.');
  const len=Number(r.headers.get('content-length')||0);if(len>FEATURE_LIMITS.maxImageBytes)throw new Error('Gambar terlalu besar.');
  const buf=Buffer.from(await r.arrayBuffer());if(!buf.length||buf.length>FEATURE_LIMITS.maxImageBytes)throw new Error('Ukuran gambar tidak didukung.');
  return {buffer:buf,type:type||'image/jpeg'};
}

  const info=platformInfo(platform);if(!info)throw new Error('Platform belum didukung.');
  const clean=String(username||'').replace(/^@/,'').trim();if(!/^[a-zA-Z0-9._-]{1,80}$/.test(clean))throw new Error('Username tidak valid.');
  const profileUrl=info.profile(clean);let profile=null;let fetchError='';
  try{profile=await fetchPublicProfile(profileUrl);}catch(e){fetchError=e.message;}
  // Search dipakai sebagai fallback, bukan sebagai sumber username acak utama.
  let results=[];if(!profile)try{results=await searchWeb(`site:${info.host} ${clean}`,5);}catch{}

function tttWinner(b){const w=[[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]];for(const[a,c,d]of w)if(b[a]&&b[a]===b[c]&&b[a]===b[d])return b[a];return b.every(Boolean)?'draw':null;}
function tttMoves(b){return b.map((v,i)=>v?null:i).filter(i=>i!==null);}
function tttMinimax(b,player,depth=0){const result=tttWinner(b);if(result==='O')return 10-depth;if(result==='X')return depth-10;if(result==='draw')return 0;const moves=tttMoves(b);if(player==='O'){let best=-Infinity;for(const i of moves){b[i]='O';best=Math.max(best,tttMinimax(b,'X',depth+1));b[i]='';}return best;}let best=Infinity;for(const i of moves){b[i]='X';best=Math.min(best,tttMinimax(b,'O',depth+1));b[i]='';}return best;}
function tttBotMove(b){let best=-Infinity,bestI=null;for(const i of tttMoves(b)){b[i]='O';const score=tttMinimax(b,'X');b[i]='';if(score>best){best=score;bestI=i;}}return bestI;}

  saveFeatureUser(chatId,{images:(getFeatureUser(chatId).images||0)+sent});if(!sent)await sendReply(bot,chatId,'Semua URL gambar ditemukan, tetapi server asal menolak pengambilan gambar. Coba query lain.',kb([[btn('🔄 Cari Lagi','feature:image'),btn('🏠 Menu','feature:menu')]]));
}

async function socialFlow(chatId,text,state){let platform=state.platform||'tiktok',username=String(text||'').trim();const m=username.match(/^(tiktok|instagram|youtube|x|twitter|facebook)\s*:\s*(.+)$/i);if(m){platform=m[1].toLowerCase();username=m[2].trim();}const d=await socialLookup(platform,username),p=d.profile,s=d.stats||{};const out=[`<b>👤 ${escHtml(d.platform)} LOOKUP</b>`,`Username: <b>@${escHtml(d.username)}</b>`,p?.title?`Nama: <b>${escHtml(p.title)}</b>`:'Nama: <i>tidak tersedia dari halaman publik</i>',p?.description?`Bio: ${escHtml(p.description.slice(0,500))}`:'Bio: <i>tidak tersedia</i>',s.followers?`Followers: <b>${escHtml(s.followers)}</b>`:'Followers: <i>tidak terbaca dari HTML publik</i>',s.following?`Following: <b>${escHtml(s.following)}</b>`:'Following: <i>tidak terbaca dari HTML publik</i>',s.likes?`Likes: <b>${escHtml(s.likes)}</b>`:'Likes: <i>tidak terbaca dari HTML publik</i>'];
  const markup=kb([[urlBtn('🔗 Buka Profil',d.profileUrl)],[btn('🖼️ Cari Foto Profil','social:image'),btn('🔄 Cari Lagi','feature:social')],[btn('🏠 Menu','feature:menu')]]);await sendReply(bot,chatId,out.join('\n'),markup);
  // Foto dikirim sebagai file Buffer, bukan URL mentah, sehingga Telegram tidak perlu mengakses server gambar sendiri.
  if(p?.image){try{await sendRemotePhoto(chatId,p.image,`<b>${escHtml(d.platform)} @${escHtml(d.username)}</b>\nFoto publik dari metadata profil.`,kb([[urlBtn('🔗 Buka Profil',d.profileUrl)]]));}catch(e){console.error('[PROFILE IMAGE]',e.message);}}
  return d;
}

async function showLeaderboard(chatId){const rows=leaderboard();if(!rows.length)return sendReply(bot,chatId,'<b>🏆 LEADERBOARD</b>\n\nBelum ada skor. Main game dulu!',kb([[btn('🎮 Game Center','feature:games'),btn('🏠 Menu','feature:menu')]]));const me=userKey(chatId);const body=rows.map(([id,u],i)=>`${i+1}. <b>${id===me?'Kamu':'Player '+id.slice(-4)}</b> — <b>${u.score||0}</b> poin • ${u.wins||0}W/${u.losses||0}L`).join('\n');await sendReply(bot,chatId,`<b>🏆 TOP 10 VICKYYVALL</b>\n\n${body}`,kb([[btn('🎮 Game Center','feature:games'),btn('🏠 Menu','feature:menu')]]));}

async function handleFeatureCallback(query){const data=String(query.data||''),chatId=String(query.message?.chat?.id||'');if(!chatId)return false;
  if(data==='feature:menu'){clearState(chatId);await sendFeatureMenu(chatId);return true;}
  if(data==='feature:games'){clearState(chatId);await sendReply(bot,chatId,'<b>🎮 GAME CENTER</b>\n\nGame nyata + skor lokal tersimpan di database.',kb(gamesKeyboard()));return true;}
  if(data==='feature:chat'){clearState(chatId);await sendReply(bot,chatId,'<b>🤖 AI CHAT</b>\n\nKirim pertanyaan seperti biasa.');return true;}
  if(data==='feature:search'){setState(chatId,{mode:'web'});await sendReply(bot,chatId,'<b>🔎 WEB SEARCH</b>\n\nKetik topik/pertanyaan. Hasil diberi tombol URL asli + ringkasan AI.',kb([[btn('🖼️ Image Search','feature:image'),btn('🏠 Menu','feature:menu')]]));return true;}
  if(data==='feature:image'){setState(chatId,{mode:'image'});await sendReply(bot,chatId,'<b>🖼️ IMAGE SEARCH</b>\n\nKetik objek/topik. Bot akan mengambil gambar publik dan mengirim file gambarnya ke Telegram, bukan cuma menampilkan URL.',kb([[btn('🔎 Web Search','feature:search'),btn('🏠 Menu','feature:menu')]]));return true;}
  if(data==='feature:social'){setState(chatId,{mode:'social',platform:'tiktok'});await sendReply(bot,chatId,'<b>👤 SOCIAL SEARCH</b>\n\nKirim username. Contoh: <code>@nama</code> atau <code>instagram:nama</code>. Data hanya dari halaman publik yang bisa diakses.',kb([[btn('TikTok','social:platform:tiktok'),btn('Instagram','social:platform:instagram')],[btn('YouTube','social:platform:youtube'),btn('X','social:platform:x')],[btn('Facebook','social:platform:facebook'),btn('🏠 Menu','feature:menu')]]));return true;}
  if(data==='feature:video'){setState(chatId,{mode:'video',platform:'all'});await sendReply(bot,chatId,'<b>🎬 VIDEO SEARCH</b>\n\nKetik judul/topik video. Bot mengembalikan hasil publik dengan tombol buka sumber.',kb([[btn('YouTube','video:platform:youtube'),btn('TikTok','video:platform:tiktok')],[btn('Semua Web','video:platform:all'),btn('🏠 Menu','feature:menu')]]));return true;}
  if(data==='feature:math'){setState(chatId,{mode:'math'});await sendReply(bot,chatId,'<b>🧮 MATH LAB</b>\n\nContoh: <code>(125*8)/5+17</code>',kb([[btn('🎮 Math Challenge','game:math'),btn('🏠 Menu','feature:menu')]]));return true;}
  if(data==='feature:media'){clearState(chatId);await sendReply(bot,chatId,'<b>📸 MEDIA LAB</b>\n\nKirim foto atau dokumen. Bot akan meneruskan media ke AI bila provider aktif dan mendukung input tersebut.',kb([[btn('🤖 AI Chat','feature:chat'),btn('🏠 Menu','feature:menu')]]));return true;}
  if(data==='feature:leaderboard'){await showLeaderboard(chatId);return true;}
  if(data==='feature:mystats'){await showMyStats(chatId);return true;}
  if(data==='feature:status'){await sendReply(bot,chatId,`<b>⚙️ SYSTEM STATUS</b>\nProvider: <code>${escHtml(activeProvider||'unset')}</code>\nModel: <code>${escHtml(activeModel||'unset')}</code>\nWeb search: <code>${process.env.WEB_SEARCH_URL?'custom endpoint':'DuckDuckGo HTML fallback'}</code>\nImage search: <code>${process.env.IMAGE_SEARCH_URL?'custom endpoint':'Bing Images fallback'}</code>\nNode: <code>${process.version}</code>`,kb([[btn('🏠 Menu','feature:menu')]]));return true;}
  if(data==='feature:about'){await sendReply(bot,chatId,'<b>VickyyVall AI</b>\n\nTelegram AI toolkit dengan fitur pencarian publik, media, game, math, statistik dan AI side-car.',kb([[btn('🏠 Menu','feature:menu')]]));return true;}
  if(data.startsWith('social:platform:')){setState(chatId,{mode:'social',platform:data.split(':').pop()});await sendReply(bot,chatId,`<b>${escHtml(platformInfo(data.split(':').pop())?.label||'Social')}</b>\nKirim username.`,kb([[btn('🏠 Menu','feature:menu')]]));return true;}
  if(data.startsWith('video:platform:')){setState(chatId,{mode:'video',platform:data.split(':').pop()});await sendReply(bot,chatId,'Kirim kata kunci video.',kb([[btn('🏠 Menu','feature:menu')]]));return true;}
  if(data==='game:quiz'){await startQuiz(chatId);return true;}
  if(data==='game:guess'){await startGuess(chatId);return true;}
  if(data==='game:ttt'){startTtt(chatId);await sendTtt(chatId);return true;}
  if(data==='game:dice'){await playDice(chatId);return true;}
  if(data==='game:math'){await startMathChallenge(chatId);return true;}
  if(data.startsWith('ttt:')){await handleTtt(chatId,Number(data.split(':')[1]));return true;}
  if(data.startsWith('quiz:')){await handleQuiz(chatId,Number(data.split(':')[1]));return true;}
  if(data==='mathgame:hint'){const g=gameState.get(chatId);if(g?.type==='mathgame'){const side=await aiSidecar(chatId,`Beri hint tanpa jawaban untuk ${g.question}.`);await sendReply(bot,chatId,escHtml(side||'Coba hitung bagian paling sederhana dulu.'),kb([[btn('Game Center','feature:games')]]));}return true;}
  if(data==='social:image'){setState(chatId,{mode:'socialImage'});await sendReply(bot,chatId,'Kirim lagi username untuk mengambil gambar profil publik dari pencarian gambar.',kb([[btn('Social Search','feature:social')]]));return true;}
  return false;
}

async function processFeatureText(msg,text){const chatId=String(msg.chat.id),state=featureState.get(chatId);if(!state)return false;
  if(await handleGuessText(chatId,text))return true;
  if(await handleMathGameText(chatId,text))return true;
  if(state.mode==='math'){try{const result=calculateMath(text);const side=await aiSidecar(chatId,`Jelaskan ${text} = ${result} dengan langkah singkat.`);await sendReply(bot,chatId,`<b>🧮 HASIL</b>\n<code>${escHtml(text)}</code> = <b>${escHtml(result)}</b>\n\n${escHtml(side||'Kalkulasi lokal berhasil.')}`,kb([[btn('🧮 Lagi','feature:math'),btn('🎮 Game Math','game:math')]]));}catch(e){await sendReply(bot,chatId,`<b>Math error:</b> ${escHtml(e.message)}`,kb([[btn('🧮 Math Lab','feature:math'),btn('🏠 Menu','feature:menu')]]));}return true;}
  if(state.mode==='image'){try{await imageSearchFlow(chatId,text);}catch(e){await sendReply(bot,chatId,`<b>Image search error:</b> ${escHtml(e.message)}`,kb([[btn('🔄 Coba Lagi','feature:image'),btn('🏠 Menu','feature:menu')]]));}return true;}
  if(state.mode==='web'||state.mode==='video'){try{const q=state.mode==='video'&&state.platform!=='all'?`${text} site:${platformInfo(state.platform)?.host||state.platform}.com`:text;const results=await searchWeb(q,FEATURE_LIMITS.maxResults);await sendSearchResults(chatId,state.mode,text,results);}catch(e){await sendReply(bot,chatId,`<b>Search error:</b> ${escHtml(e.message)}`,kb([[btn('🔄 Coba Lagi',state.mode==='video'?'feature:video':'feature:search'),btn('🏠 Menu','feature:menu')]]));}return true;}
  if(state.mode==='social'){try{await socialFlow(chatId,text,state);}catch(e){await sendReply(bot,chatId,`<b>Social error:</b> ${escHtml(e.message)}`,kb([[btn('🔄 Coba Lagi','feature:social'),btn('🏠 Menu','feature:menu')]]));}return true;}
  if(state.mode==='socialImage'){try{const m=String(text).trim().match(/^(tiktok|instagram|youtube|x|twitter|facebook)\s*:\s*(.+)$/i);const platform=m?.[1]||'tiktok',username=m?.[2]||text;const urls=await searchImages(`${platform} ${username} profile`,3);let sent=0;for(const u of urls){try{await sendRemotePhoto(chatId,u,`<b>${escHtml(platform)} @${escHtml(username.replace(/^@/,''))}</b>`,kb([[btn('🔄 Cari Lagi','feature:social'),btn('🏠 Menu','feature:menu')]]));sent++;}catch(e){console.error('[SOCIAL IMAGE]',e.message);}}if(!sent)await sendReply(bot,chatId,'Gambar profil publik tidak bisa diambil dari sumber yang ditemukan.');}catch(e){await sendReply(bot,chatId,`<b>Image error:</b> ${escHtml(e.message)}`);}return true;}
  return false;
}

bot.on('callback_query',async q=>{try{await handleFeatureCallback(q);}catch(e){console.error('[FEATURE CALLBACK]',e.message);}});
bot.onText(/^\/(menu|features|games|math|search|social|video)(?:@\w+)?(?:\s+(.+))?$/i,async(msg,match)=>{const c=String(match?.[1]||'').toLowerCase(),a=cleanText(match?.[2]||''),id=String(msg.chat.id);if(c==='menu'||c==='features')return sendFeatureMenu(id);if(c==='games')return sendReply(bot,id,'<b>🎮 GAME CENTER</b>',kb(gamesKeyboard()));if(c==='math'){setState(id,{mode:'math'});return a?processFeatureText(msg,a):sendReply(bot,id,'Kirim ekspresi matematika.',kb([[btn('🏠 Menu','feature:menu')]]));}if(c==='search'){setState(id,{mode:'web'});return a?processFeatureText(msg,a):sendReply(bot,id,'Ketik topik pencarian.',kb([[btn('🏠 Menu','feature:menu')]]));}if(c==='social'){setState(id,{mode:'social',platform:'tiktok'});return a?processFeatureText(msg,a):sendReply(bot,id,'Ketik username akun publik.',kb([[btn('Instagram','social:platform:instagram'),btn('YouTube','social:platform:youtube')],[btn('🏠 Menu','feature:menu')]]));}if(c==='video'){setState(id,{mode:'video',platform:'all'});return a?processFeatureText(msg,a):sendReply(bot,id,'Ketik judul/kata kunci video.',kb([[btn('YouTube','video:platform:youtube'),btn('TikTok','video:platform:tiktok')],[btn('🏠 Menu','feature:menu')]]));}});
